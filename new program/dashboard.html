⚡ Supabase Messages Table Setup

Run this query in your Supabase SQL Editor to enable cloud storage & realtime.

CREATE TABLE IF NOT EXISTS public.messages (
  id TEXT PRIMARY KEY,
  sender_email TEXT,
  sender_name TEXT,
  sender_role TEXT,
  recipient_email TEXT,
  recipient_name TEXT,
  recipient_role TEXT,
  content TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  read BOOLEAN DEFAULT FALSE
);

ALTER TABLE public.messages ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow public select messages" ON public.messages FOR SELECT USING (true);
CREATE POLICY "Allow public insert messages" ON public.messages FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow public update messages" ON public.messages FOR UPDATE USING (true);

ALTER PUBLICATION supabase_realtime ADD TABLE public.messages;
Messages work seamlessly with instant multi-tab sync now, and will sync across devices once this table exists in Supabase.
